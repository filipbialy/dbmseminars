  <script>
    const decimalToBinaryButton = document.getElementById('decimal-to-binary-btn') as HTMLButtonElement;
const decimalInput = document.getElementById('decimal-input') as HTMLInputElement;
const resultParagraph = document.getElementById('result-p') as HTMLParagraphElement;
const errorParagraph = document.getElementById('error-p') as HTMLParagraphElement;

decimalToBinaryButton.addEventListener('click', () => {
    const inputValue = Number(decimalInput.value);
    const result = decimalToBinary(inputValue);
    errorParagraph.innerText = '';
    resultParagraph.innerText = '';
    resultParagraph.innerText = `${inputValue} converted to binary is: ${result}`;
});

decimalInput.addEventListener('input', (event) => {
    const input = (event.target as HTMLInputElement).value;
    if ((input?.length || 0) === 0) {
        decimalToBinaryButton.setAttribute('disabled', '');
        return;
    }
    decimalToBinaryButton.removeAttribute('disabled');
}); 

window.addEventListener('error', (err) => {
    errorParagraph.innerText = err.message ?? 'An error has occurred, please check your input';
});



const validateDecimal = (decimalNumber: number) : number => {

      if(Number.isNaN(decimalNumber)) {
      throw new Error("Please insert a valid number");
   }

    if(!Number.isInteger(decimalNumber)) {
        throw new Error("Please enter a decimal");
    }

       if(decimalNumber < 0) {
        throw new Error("Decimal number must be positive");
    }
    
    return decimalNumber;
}



const decimalToBinary = (input: any) : string => {
    const validatedDecimal = validateDecimal(input);

    let decimalCopy = validatedDecimal;
    let result = '';

    while(decimalCopy > 0) {
        const rest = decimalCopy % 2;
        result = rest + result; 
        decimalCopy = Math.floor(decimalCopy / 2);
    }

    return result;
}
